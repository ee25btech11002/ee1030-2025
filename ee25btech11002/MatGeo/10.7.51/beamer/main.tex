\begin{frame}{Question}
Find the coordinates of all the points $\vec{P}$ on the ellipse $\frac{x^2}{a^2}+\frac{y^2}{b^2}$=1, for which the area of the triangle $PON$ is maximum, where $\vec{O}$ denotes the origin and $\vec{N}$, the foot of the perpendicular from $\vec{O}$ to the tangent at $\vec{P}$
\end{frame}

\begin{frame}{Solution}
The equation of the ellipse is given as
\begin{align}
\vec{x}^\top\vec{V}\vec{x} + 2\vec{u}^\top\vec{x} + f &= 0\\
\vec{V} = \myvec{\frac{1}{a^2} & 0\\0 & \frac{1}{b^2}},\vec{u} &= \myvec{0\\0} , f = -1\\
\vec{x}^\top\vec{V} \vec{x} &= 1
\end{align}
The equation of a tangent to $\vec{P}$ is 
\begin{align}
    \brak{\vec{V}\vec{P} + \vec{u}}^\top \vec{x} + \vec{u}^\top\vec{P} + f = 0\\
    \brak{\vec{V}\vec{P}}^\top\vec{x} + f = 0
\end{align}
As $\vec{N}$ is the foot of the perpendicular from $\vec{O}$ to the tangent at $\vec{P}$
    \begin{align}
    \vec{N} &= \vec{0} - \frac{\brak{\vec{VP}}^T \vec{0} + f}{\norm{\vec{VP}}} \, \vec{VP}\\
    \vec{N} &= \frac{\vec{VP}}{\norm{\vec{VP}}}
\end{align} 
\end{frame}



\begin{frame}{Solution}
The cross product of two 2D vectors $\vec{x}$ and 
$\vec{y}$  can be written in matrix form using the skew-symmetric matrix

\begin{align}
    \vec{J} = \myvec{0 & 1 \\ -1 & 0}.
\end{align}

Hence,
\begin{align}
    \vec{x} \times \vec{y} = \vec{x}^\top \vec{J} \vec{y}.
\end{align}

The area of the triangle $\triangle PON$ is given by
\begin{align}
    \text{Area}\brak{PON}
    = \frac{1}{2} \frac{\abs{\vec{P}^\top \vec{J} \vec{V} \vec{P}}}{\norm{\vec{V}\vec{P}}^2},
\end{align}
\end{frame}


\begin{frame}{Solution}
Let
\begin{align}
    \vec{Q} &= \vec{V}\vec{P} \\
    \vec{P} &= \vec{V}^{-1}\vec{Q}.
\end{align}

Then, the area becomes
\begin{align}
    \text{Area}\brak{PON}
    &= \frac{1}{2} \frac{\abs{\brak{\vec{V}^{-1}\vec{Q}}^\top \vec{J} \vec{Q}}}{\norm{\vec{Q}}^2} \\
    &= \frac{1}{2} \frac{\abs{\vec{Q}^\top \brak{\vec{V}^{-1}\vec{J}} \vec{Q}}}{\norm{\vec{Q}}^2}.
\end{align}

Since $\vec{Q}^\top \vec{M} \vec{Q}$ depends only on the symmetric part of $\vec{M}$, define
\begin{align}
    \vec{S} = \frac{\vec{V}^{-1}\vec{J} + \brak{\vec{V}^{-1}\vec{J}}^\top}{2}
\end{align}
\end{frame}


\begin{frame}{Solution}
Hence,
\begin{align}
    \text{Area}\brak{\vec{Q}} = \frac{1}{2}\frac{\abs{\vec{Q}^\top \vec{S} \vec{Q}}}{\norm{\vec{Q}}^2}.
\end{align}

Now,
\begin{align}
    \vec{V}^{-1} = \myvec{a^2 & 0 \\ 0 & b^2}, \quad 
    \vec{J} = \myvec{0 & 1 \\ -1 & 0}.
\end{align}
Thus,
\begin{align}
    \vec{V}^{-1}\vec{J} = \myvec{0 & a^2 \\ -b^2 & 0}, \quad
    \brak{\vec{V}^{-1}\vec{J}}^\top = \myvec{0 & -b^2 \\ a^2 & 0}.
\end{align}
\end{frame}


\begin{frame}{Solution}
So,
\begin{align}
    \vec{S} 
    &= \frac{1}{2}\brak{\vec{V}^{-1}\vec{J} + \brak{\vec{V}^{-1}\vec{J}}^\top} \\
    &= \frac{1}{2}\myvec{0 & a^2 - b^2 \\ a^2 - b^2 & 0} \\
    &= \frac{a^2 - b^2}{2}\myvec{0 & 1 \\ 1 & 0}.
\end{align}

The eigenvalues and eigenvectors of 
$\myvec{0 & 1 \\ 1 & 0}$ are
\begin{align}
    \lambda_1 = 1, \ \vec{v}_1 = \myvec{1 \\ 1}, \quad
    \lambda_2 = -1, \ \vec{v}_2 = \myvec{1 \\ -1}.
\end{align}
\end{frame}


\begin{frame}{Solution}
Hence, the eigenvalues of $\vec{S}$ are $\pm \frac{a^2 - b^2}{2}$, and the maximum absolute eigenvalue is
\begin{align}
    \lambda_{\max} = \frac{\abs{a^2 - b^2}}{2}.
\end{align}

The maximum area is therefore
\begin{align}
    \text{Area}_{\max} = \frac{\abs{a^2 - b^2}}{4}
\end{align}

The corresponding eigenvector is
\begin{align}
    \vec{Q} \propto \myvec{1 \\ 1}.
\end{align}Hence,
\begin{align}
    \vec{P} = \vec{V}^{-1}\vec{Q} \propto \myvec{a^2 \\ b^2}.
\end{align}

Imposing the ellipse condition $\vec{P}^\top \vec{V} \vec{P} = 1$,
\begin{align}
    1 = \lambda^2 \brak{a^2 + b^2}
    \implies \lambda = \pm \frac{1}{\sqrt{a^2 + b^2}}.
\end{align}

Thus, the required points are
\begin{align}
    \vec{P} = \pm \frac{1}{\sqrt{a^2 + b^2}}\myvec{a^2 \\ b^2}
\end{align}
and the maximum area of $\triangle PON$ is
\begin{align}
    \text{Area}_{\max} = \frac{\abs{a^2 - b^2}}{4}
\end{align}
\end{frame}

typedef struct {

// ---------- Compute V @ P ----------
void tangent_normal_at(double a, double b, Vec2 P, Vec2 *n) {
    n->x = P.x / (a * a);
    n->y = P.y / (b * b);
}

// ---------- Foot of perpendicular from origin to line n·x = 1 ----------
void foot_from_origin_to_line(Vec2 n, double c, Vec2 *N) {
    double denom = n.x * n.x + n.y * n.y;
    N->x = (c / denom) * n.x;
    N->y = (c / denom) * n.y;
}
// ---------- Cross product area ----------
double area_triangle(Vec2 P, Vec2 O, Vec2 N) {
    double cross = P.x * N.y - P.y * N.x;
    return 0.5 * fabs(cross);
}
// ---------- Compute optimal P points ----------
void compute_P_points(double a, double b, Vec2 *P_plus, Vec2 *P_minus) {
    double lam = 1.0 / sqrt(a * a + b * b);
    P_plus->x = lam * a * a;
    P_plus->y = lam * b * b;
    \end{lstlisting}

// ---------- Compute tangent line points (parametric) ----------
void tangent_line_points(Vec2 n, Vec2 N, double span, int npts, double *x_out, double *y_out) {
    // J @ n = (n.y, -n.x)
    double tx = n.y;
    double ty = -n.x;
    double step = (2 * span) / (npts - 1);
    for (int i = 0; i < npts; i++) {
        double s = -span + i * step;
        x_out[i] = N.x + tx * s;
        y_out[i] = N.y + ty * s;
    }
}
