\begin{frame}{Question}
In a linear inverse problem, the coefficient matrix, $\vec{A} = \myvec{ 2.00 & 2.01 \\ 2.01 & 2.00 }$.
 The eigenvalues of $\vec{A}$ are
  \begin{enumerate}
   \begin{multicols}{4}
   \item $\brak{4.01\,, -0.01}$
   \item $\brak{-4.01\,,-0.01}$
   \item $\brak{4.01\,,0.01}$
   \item $\brak{-4.01\,,1.01}$
   \end{multicols}
   \end{enumerate}
\end{frame}

\begin{frame}{Solution:}
The characteristic equation for matrix $\vec{A}$ is given by
\begin{align}
    \mydet{\vec{A} - \lambda\vec{I}} = 0
\end{align}
\begin{align}
    \mydet{2.00-\lambda & 2.01\\ 2.01 & 2.00-\lambda} = 0\\
     \brak{2.00-\lambda}^2 - \brak{2.01}^2 = 0\\
     \brak{\brak{2.00-\lambda} - 2.01}\brak{\brak{2.00-\lambda} + 2.01} = 0\\
     \brak{-\lambda - 0.01}\brak{-\lambda + 4.01} = 0\\
     \therefore \lambda_1 = 4.01,\quad \lambda_2 = -0.01\\
\end{align}
\end{frame}

\begin{frame}{Solution:}
Hence, the eigenvalues are
\begin{align}
    \lambda_1 = 4.01,\ \lambda_2 = -0.01
\end{align}
\end{frame}
