\begin{frame}{Question}
The lower triangular matrix $\vec{L}$ in the LU factorization of the matrix
$\myvec{
    25&5&4\\10&8&16\\8&10&22
}$ is written as $\myvec{
    1&0&0\\L_{21}&1&0\\L_{31}&L_{32}&1
}$. The element $L_{32}$ is 
\begin{enumerate}
\begin{multicols}{4}
\item 1.0 
\item 1.4 
\item 0.4 
\item 0.32
\end{multicols}
\end{enumerate}
\end{frame}

\begin{frame}{Solution}
Let 
\begin{align}
    \vec{A} &= \myvec{
        25 & 5 & 4\\
        10 & 8 & 16\\
        8 & 10 & 22
    }
\end{align}
The lower triangular and upper triangular matrices are written as
\begin{align}
    \vec{L} &= \myvec{
        1 & 0 & 0\\
        L_{21} & 1 & 0\\
        L_{31} & L_{32} & 1
    }, \quad
    \vec{U} = \myvec{
        u_{11} & u_{12} & u_{13}\\
        0 & u_{22} & u_{23}\\
        0 & 0 & u_{33}
    }
\end{align}
and
\begin{align}
    \vec{A} = \vec{LU}
\end{align}
\begin{align}
    a_{ij} &= \sum_{k=1}^{\min\brak{i,j}} L_{ik}U_{kj}
\end{align}
\end{frame}



\begin{frame}{Solution}
Using Doolittle-style relations \brak{\text{and the elementwise formula}}:

\begin{align}
u_{11} &= a_{11} = 25\\
u_{12} &= a_{12} = 5\quad u_{13}=a_{13}=4\\
L_{21} &= \frac{a_{21}}{u_{11}}=\frac{10}{25}=0.4\\
L_{31} &= \frac{a_{31}}{u_{11}}=\frac{8}{25}=0.32
\end{align}
\begin{align}
u_{22} &= a_{22} - L_{21}u_{12} = 8 - \brak{0.4}\brak{5} = 8 - 2 = 6\\
u_{23} &= a_{23} - L_{21}u_{13} = 16 - \brak{0.4}\brak{4} = 16 - 1.6 = 14.4
\end{align}
\end{frame}

\begin{frame}{Solution}
Now use the formula for $a_{32}$ $\brak{\text{with   min\brak{3,2} = 2}}$
\begin{align}
a_{32} = L_{31}u_{12} + L_{32}u_{22}
\end{align}
Solve for $L_{32}$:
\begin{align}
L_{32} &= \frac{a_{32} - L_{31}u_{12}}{u_{22}}
= \frac{10 - \brak{0.32}\brak{5}}{6}
= \frac{10 - 1.6}{6}
= \frac{8.4}{6}
= 1.4
\end{align}

\begin{align}
L_{32} = 1.4
\end{align}
\end{frame}
