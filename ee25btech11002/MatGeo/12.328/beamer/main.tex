\begin{frame}{Question}
For the given matrix $$\vec{Q}=\myvec{\frac{1}{\sqrt{2}}&0&\frac{1}{\sqrt{2}}\\
0&1&0\\
-\frac{1}{\sqrt{2}}&0&\frac{1}{\sqrt{2}}},$$ which of the following statements is/are true?
\begin{enumerate}
	\begin{multicols}{2}
    \item $\vec{Q}$ is an orthogonal matrix
    \item $\vec{Q}^{\top}=\vec{Q}^{-1}$
    \item $\vec{Q}$ is a singular matrix
    \item $\vec{Q}$ is a symmetric matrix
    \end{multicols}
\end{enumerate}
\end{frame}

\begin{frame}{Theoretical Solution}
\begin{align}
\vec{Q}^\top &=
\myvec{
\frac{1}{\sqrt{2}} & 0 & -\frac{1}{\sqrt{2}}\\
0 & 1 & 0\\
\frac{1}{\sqrt{2}} & 0 & \frac{1}{\sqrt{2}}
}
\\
 \vec{Q}^\top \vec{Q} &=
\myvec{
\frac{1}{\sqrt{2}} & 0 & -\frac{1}{\sqrt{2}}\\
0 & 1 & 0\\
\frac{1}{\sqrt{2}} & 0 & \frac{1}{\sqrt{2}}
}
\myvec{
\frac{1}{\sqrt{2}} & 0 & \frac{1}{\sqrt{2}}\\
0 & 1 & 0\\
-\frac{1}{\sqrt{2}} & 0 & \frac{1}{\sqrt{2}}
}\\
\vec{Q}^\top \vec{Q} &=
\myvec{
1 & 0 & 0\\
0 & 1 & 0\\
0 & 0 & 1
}
= \vec{I}
\end{align}
\end{frame}



\begin{frame}{Theoretical Solution}
\begin{align}
\det\brak{\vec{Q}} &= 1 \neq 0
\end{align}
Also,
\begin{align}
\vec{Q}^\top \neq \vec{Q}
\end{align}
Thereby, $\vec{Q}$ is an orthogonal matrix i.e. $\vec{Q}^{\top}=\vec{Q}^{-1}$
\end{frame}
