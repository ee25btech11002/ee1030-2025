\begin{frame}{Question}
The tangent plane to the surface $x^2 + y^2 + z = 9$ at the point $(1, 2, 4)$ is
\begin{enumerate}
        \item $2x+4y+z=14$
        \item $4x+2y+z=12$
        \item $x+4y+2z=17$
        \item $4x+y+2z=14$
\end{enumerate}
\end{frame}

\begin{frame}{Theoretical Solution}
The equation of the surface is\\
\begin{align}
\vec{x}^\top V \vec{x} + 2u^\top \vec{x} + f = 0
\end{align}

\begin{align}
\vec{x} = \myvec{x\\y\\z}, \quad 
\vec{V} = \myvec{1&0&0\\0&1&0\\0&0&0}, \quad
\vec{u} = \myvec{0\\0\\\frac{1}{2}}, \quad
f = -9
\end{align}

The tangent plane is given as
\begin{align}
    \brak{\vec{Vq} + \vec{u}}^\top \vec{x} + \vec{u}^\top \vec{q} + f = 0
\end{align}
\end{frame}

\begin{frame}{Theoretical Solution}
where
\begin{align}
    \vec{q} = \myvec{1\\2\\4}
\end{align}

\begin{align}
    \brak{\myvec{1&0&0\\0&1&0\\0&0&0}\myvec{1\\2\\4} + \myvec{0\\0\\\frac{1}{2}}}^\top \vec{x} +
    \myvec{0\\0\\\frac{1}{2}}^\top \myvec{1\\2\\4} = 9
\end{align}
\begin{align}
    \myvec{1\\2\\\frac{1}{2}}^\top \vec{x} = 7
\end{align}
\begin{align}
    \implies 2x + 4y + z = 14
\end{align}
\end{frame}

#include <stdio.h>
#include <math.h>

// Function to compute Z values for the surface: z = 9 - x^2 - y^2
void compute_surface(int n, double *x, double *y, double **Z) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            Z[i][j] = 9.0 - x[i]*x[i] - y[j]*y[j];
        }
    }
}
    \end{lstlisting}
\end{frame}

// Function to compute Z values for the tangent plane: z = 14 - 2x - 4y
void compute_plane(int n, double *x, double *y, double **Z) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            Z[i][j] = 14.0 - 2.0*x[i] - 4.0*y[j];
        }
    }
}
    \end{lstlisting}
